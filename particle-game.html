const c = document.getElementById("game"); const ctx = c.getContext("2d"); // ---------------- AUDIO ---------------- const bgMusic = new Audio("assets/untitled.mp3"); bgMusic.loop = true; bgMusic.volume = 0.4; const catchSound = new Audio("assets/hit.mp3"); catchSound.volume = 0.7; // ---------------- GAME STATE ---------------- let player, particles, score; let time = 30; let running = false; let targetX = 350; // ---------------- INPUT ---------------- document.addEventListener("mousemove", function(e){ let rect = c.getBoundingClientRect(); targetX = e.clientX - rect.left - player.w/2; }); // ---------------- INIT ---------------- function reset(){ player = {x:350,y:450,w:100,h:20}; particles = []; score = {u:0,d:0,e:0}; document.getElementById("u").innerText=0; document.getElementById("d").innerText=0; document.getElementById("e").innerText=0; } // ---------------- SPAWN ---------------- function spawn(){ const types=["u","d","e"]; particles.push({ x:Math.random()*760, y:0, t:types[Math.floor(Math.random()*3)], s:2+Math.random()*3 }); } // ---------------- UPDATE ---------------- function update(){ player.x += (targetX - player.x) * 0.2; player.x = Math.max(0, Math.min(700, player.x)); for(let i=particles.length-1;i>=0;i--){ let p=particles[i]; p.y+=p.s; // collision if( p.x>player.x && p.xplayer.y ){ score[p.t]++; document.getElementById(p.t).innerText=score[p.t]; catchSound.currentTime = 0; catchSound.play(); particles.splice(i,1); } if(p.y>500) particles.splice(i,1); } } // ---------------- DRAW ---------------- function draw(){ ctx.clearRect(0,0,800,500); ctx.fillStyle="white"; ctx.fillRect(player.x,player.y,player.w,player.h); particles.forEach(p=>{ ctx.fillStyle = p.t==="u"?"blue": p.t==="d"?"green":"orange"; ctx.beginPath(); ctx.arc(p.x,p.y,15,0,Math.PI*2); ctx.fill(); ctx.fillStyle="white"; ctx.fillText(p.t,p.x-5,p.y+5); }); } // ---------------- START ---------------- function startGame(){ document.getElementById("menu").style.display="none"; document.getElementById("hud").style.display="block"; reset(); running=true; // 🎵 START MUSIC (important user gesture rule) bgMusic.currentTime = 0; bgMusic.play(); let gameLoop = setInterval(()=>{ if(running){update();draw();} },16); let spawnLoop = setInterval(()=>{ if(running) spawn(); },700); let timerLoop = setInterval(()=>{ if(running){ time--; document.getElementById("timer").innerText=time; if(time<=0){ running=false; bgMusic.pause(); clearInterval(gameLoop); clearInterval(spawnLoop); clearInterval(timerLoop); document.getElementById("hud").style.display="none"; document.getElementById("over").style.display="block"; document.getElementById("score").innerHTML= `u=${score.u}
d=${score.d}
e=${score.e}`; } } },1000); }
